## Hardware Profile Detection

```typescript
// packages/coordination/src/optimization/connection-pool.ts

import * as os from 'os';
import { MqttClient } from 'mqtt';

export interface HardwareProfile {
  maxConnections: number;
  healthCheckInterval: number;
  idleTimeout: number;
}

export const HARDWARE_PROFILES: Record<string, HardwareProfile> = {
  'pi-2b': { maxConnections: 3, healthCheckInterval: 30000, idleTimeout: 120000 },
  'pi-5': { maxConnections: 5, healthCheckInterval: 30000, idleTimeout: 120000 },
  'beelink': { maxConnections: 10, healthCheckInterval: 30000, idleTimeout: 120000 },
  'default': { maxConnections: 5, healthCheckInterval: 30000, idleTimeout: 120000 }
};

export function detectHardwareProfile(): HardwareProfile {
  const cpus = os.cpus();
  const totalMem = os.totalmem();
  const model = cpus[0]?.model || '';

  // Raspberry Pi 2B: ARMv7, 1GB RAM
  if (model.includes('ARMv7') && totalMem < 2 * 1024 * 1024 * 1024) {
    return HARDWARE_PROFILES['pi-2b'];
  }
  // Raspberry Pi 5: ARMv8, 4-8GB RAM
  if (model.includes('ARMv8') && totalMem >= 2 * 1024 * 1024 * 1024) {
    return HARDWARE_PROFILES['pi-5'];
  }
  // Beelink/PC: x86_64, >8GB RAM
  if (process.platform === 'linux' && model.includes('Intel') && totalMem > 8 * 1024 * 1024 * 1024) {
    return HARDWARE_PROFILES['beelink'];
  }

  return HARDWARE_PROFILES['default'];
}

export interface PoolConfig {
  brokerUrl: string;
  options?: any;
  profile?: HardwareProfile;
}

export class MqttConnectionPool {
  private connections: Map<string, MqttClient> = new Map();
  private lastUsed: Map<string, number> = new Map();
  private config: PoolConfig;
  private profile: HardwareProfile;
  private healthCheckTimer?: NodeJS.Timeout;

  constructor(config: PoolConfig) {
    this.config = config;
    this.profile = config.profile || detectHardwareProfile();
    this.startHealthCheck();
  }

  async acquire(): Promise<MqttClient> {
    // Check for existing healthy connection
    for (const [id, client] of this.connections) {
      if (this.isHealthy(client)) {
        this.lastUsed.set(id, Date.now());
        return client;
      }
    }

    // Create new connection if under limit
    if (this.connections.size < this.profile.maxConnections) {
      const client = await this.createConnection();
      const id = this.generateConnectionId();
      this.connections.set(id, client);
      this.lastUsed.set(id, Date.now());
      return client;
    }

    // Evict least recently used connection
    const lruId = [...this.lastUsed.entries()]
      .sort((a, b) => a[1] - b[1])[0][0];
    await this.release(lruId);

    // Create new connection after eviction
    const client = await this.createConnection();
    const id = this.generateConnectionId();
    this.connections.set(id, client);
    this.lastUsed.set(id, Date.now());
    return client;
  }

  async release(connectionId: string): Promise<void> {
    const client = this.connections.get(connectionId);
    if (!client) return;

    // Check if idle beyond timeout
    const lastUsed = this.lastUsed.get(connectionId) || 0;
    if (Date.now() - lastUsed > this.profile.idleTimeout) {
      await this.remove(connectionId);
    }
  }

  private async createConnection(): Promise<MqttClient> {
    return new Promise((resolve, reject) => {
      const client = mqtt.connect(this.config.brokerUrl, this.config.options);

      client.on('connect', () => resolve(client));
      client.on('error', (err) => reject(err));
    });
  }

  private isHealthy(client: MqttClient): boolean {
    // Check if client is connected and not in error state
    return client.connected && !client.stream.destroyed;
  }

  private async remove(connectionId: string): Promise<void> {
    const client = this.connections.get(connectionId);
    if (client) {
      client.end();
      this.connections.delete(connectionId);
      this.lastUsed.delete(connectionId);
    }
  }

  private generateConnectionId(): string {
    return `conn-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }

  private startHealthCheck(): void {
    this.healthCheckTimer = setInterval(() => {
      const now = Date.now();

      for (const [id, client] of this.connections) {
        // Check idle timeout
        const lastUsed = this.lastUsed.get(id) || 0;
        if (now - lastUsed > this.profile.idleTimeout) {
          this.remove(id);
          continue;
        }

        // Validate connection health
        if (!this.isHealthy(client)) {
          this.remove(id);
        }
      }
    }, this.profile.healthCheckInterval);
  }

  async stop(): Promise<void> {
    if (this.healthCheckTimer) {
      clearInterval(this.healthCheckTimer);
    }

    // Close all connections
    for (const [id, client] of this.connections) {
      await this.remove(id);
    }
  }

  getStats() {
    return {
      totalConnections: this.connections.size,
      maxConnections: this.profile.maxConnections,
      healthyConnections: [...this.connections.values()].filter(c => this.isHealthy(c)).length
    };
  }
}
```

## Key Requirements

1. Hardware detection via os.cpus() and os.totalmem():
   - Pi 2B: maxConnections=3 (1GB RAM, ARMv7)
   - Pi 5: maxConnections=5 (4-8GB RAM, ARMv8)
   - Beelink: maxConnections=10 (>8GB RAM, x86_64)
   - Default: maxConnections=5

2. LRU eviction: When pool at max capacity, evict least recently used connection before creating new one

3. Health checks: Every 30 seconds (healthCheckInterval)
   - Remove idle connections (>2 minutes unused)
   - Remove unhealthy connections (disconnected or stream destroyed)

4. Fallback on pool exhaustion: If all connections unhealthy, create new connection anyway (temporary exceed limit)
