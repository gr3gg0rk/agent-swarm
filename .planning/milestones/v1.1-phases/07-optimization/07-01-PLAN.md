```typescript
// packages/coordination/src/optimization/batcher.ts

import { MessageEnvelope } from '../communication/message';

export interface BatchConfig {
  tasks: { windowMs: number; maxSize: number };
  status: { windowMs: number; maxSize: number };
  heartbeats: { windowMs: number; maxSize: number };
}

export const DEFAULT_BATCH_CONFIG: BatchConfig = {
  tasks: { windowMs: 10, maxSize: 50 },      // 10ms OR 50 messages
  status: { windowMs: 50, maxSize: 100 },    // 50ms OR 100 messages
  heartbeats: { windowMs: 100, maxSize: 20 } // 100ms OR 20 messages
};

export class MessageBatcher {
  private buffers: Map<string, MessageEnvelope[]> = new Map();
  private timers: Map<string, NodeJS.Timeout> = new Map();
  private mqttClient: any; // MqttClient interface
  private config: BatchConfig;

  constructor(mqttClient: any, config: BatchConfig = DEFAULT_BATCH_CONFIG) {
    this.mqttClient = mqttClient;
    this.config = config;
  }

  async publish(topic: string, envelope: MessageEnvelope): Promise<void> {
    const type = this.getMessageType(envelope);

    // CRITICAL: Don't batch task assignments (latency critical)
    if (type === 'task') {
      await this.mqttClient.publish(topic, envelope);
      return;
    }

    // Add to buffer
    if (!this.buffers.has(type)) {
      this.buffers.set(type, []);
      this.scheduleFlush(type, this.config[type].windowMs);
    }
    this.buffers.get(type)!.push(envelope);

    // Flush if size limit reached (dual-trigger)
    if (this.buffers.get(type)!.length >= this.config[type].maxSize) {
      await this.flush(type);
    }
  }

  private async flush(type: string): Promise<void> {
    const messages = this.buffers.get(type) || [];
    if (messages.length === 0) return;

    // Clear buffer and timer
    this.buffers.set(type, []);
    const timer = this.timers.get(type);
    if (timer) {
      clearTimeout(timer);
      this.timers.delete(type);
    }

    // Publish batch as single MessagePack array
    const { MessagePack } = require('msgpackr');
    const payload = MessagePack.encode(messages);
    await this.mqttClient.publish(this.topicFor(type), { payload });
  }

  private scheduleFlush(type: string, windowMs: number): void {
    const timer = setTimeout(() => this.flush(type), windowMs);
    this.timers.set(type, timer);
  }

  private topicFor(type: string): string {
    // Batch topics: swarm/batch/{type}
    return `swarm/batch/${type}`;
  }

  private getMessageType(envelope: MessageEnvelope): string {
    // Map envelope.type to batch config key
    const type = envelope.type;
    if (type === 'task' || type === 'result' || type === 'cancel') return 'tasks';
    if (type === 'progress') return 'status';
    if (type === 'heartbeat' || type === 'load_metrics') return 'heartbeats';
    return 'status'; // default
  }

  async stop(): Promise<void> {
    // Flush all pending buffers on shutdown
    for (const type of this.buffers.keys()) {
      await this.flush(type);
    }
  }
}
```

## Key Requirements

1. Dual-trigger flushing: Flush on EITHER time threshold OR buffer size limit (whichever first)
2. Don't batch 'task' type messages - latency matters more than throughput for task assignments
3. Use MessagePack for batch payload serialization (already in project)
4. Batch topic pattern: `swarm/batch/{type}` where type is 'tasks', 'status', or 'heartbeats'
5. Graceful shutdown: flush all pending buffers in stop() method

## Degradation Mode

Wrap publish() in try-catch. If batcher fails, fall back to direct MqttClient.publish():

```typescript
async publish(topic: string, envelope: MessageEnvelope): Promise<void> {
  try {
    // ... batching logic
  } catch (error) {
    // Fallback to direct publish
    await this.mqttClient.publish(topic, envelope);
  }
}
```
