    - Create indexes: idx_agent_status, idx_last_heartbeat
    - Create project_context table: key (TEXT PRIMARY KEY), value (TEXT), updated_at (INTEGER)
    - Create tasks_archive table with archived_at (INTEGER)
    - Create status_archive table with id (INTEGER PRIMARY KEY AUTOINCREMENT)
    - Create indexes on archive tables
