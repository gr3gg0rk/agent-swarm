    Export interface HealthStatus {
      status: 'healthy' | 'unhealthy'
      agentId: string
      timestamp: string
      checks: {
        database: 'connected' | 'disconnected' | 'skipped'
        mqtt: 'connected' | 'disconnected' | 'skipped'
        heartbeat: 'publishing' | 'stopped' | 'skipped'
      }
    }

    Export class HealthCheckServer {
      private server?: ReturnType<typeof createServer>
      private lastHeartbeatTime?: number

      constructor(private config: HealthCheckConfig) {
        // Track last heartbeat time if publisher provided
        if (config.heartbeatPublisher) {
          this.setupHeartbeatTracking(config.heartbeatPublisher)
        }
      }

      start(): void {
        this.server = createServer((req, res) => this.handleRequest(req, res))
        this.server.listen(this.config.port, () => {
          logger.info({ port: this.config.port, agentId: this.config.agentId }, 'Health check server started')
        })
      }

      stop(): Promise<void> {
        return new Promise((resolve) => {
          if (this.server) {
            this.server.close(() => {
              logger.info('Health check server stopped')
              resolve()
            })
          } else {
            resolve()
          }
        })
      }

      private setupHeartbeatTracking(publisher: HeartbeatPublisher): void {
        // Monitor heartbeat publishing - update lastHeartbeatTime
        // Note: This would require HeartbeatPublisher to emit events or we track externally
        // For now, we'll check if publisher is started (has active interval)
      }

      private async handleRequest(req: IncomingMessage, res: ServerResponse): Promise<void> {
        if (req.url === '/health' && req.method === 'GET') {
          const healthStatus = await this.getHealthStatus()
          const statusCode = healthStatus.status === 'healthy' ? 200 : 503

          res.writeHead(statusCode, { 'Content-Type': 'application/json' })
          res.end(JSON.stringify(healthStatus))
        } else {
          res.writeHead(404)
          res.end('Not Found')
        }
      }

      private async getHealthStatus(): Promise<HealthStatus> {
        const checks: HealthStatus['checks'] = {
          database: 'skipped',
          mqtt: 'skipped',
          heartbeat: 'skipped'
        }

        let isHealthy = true

        // Check database connectivity
        if (this.config.database) {
          try {
            const row = this.config.database.prepare('SELECT 1 AS test').get()
            checks.database = (row && (row as any).test === 1) ? 'connected' : 'disconnected'
            if (checks.database === 'disconnected') isHealthy = false
          } catch (error) {
            checks.database = 'disconnected'
            isHealthy = false
          }
        }

        // Check MQTT connectivity
        if (this.config.mqttClient) {
          // Check if client is connected (implementation depends on MQTT wrapper)
          // Assuming MqttClientMinimal has a way to check connection status
          checks.mqtt = 'connected' // Placeholder - actual implementation depends on mqtt.ts
          // If mqtt.ts doesn't expose connection status, skip this check
        }

        // Check heartbeat publisher is running
        if (this.config.heartbeatPublisher) {
          // Verify heartbeats are being published (check last heartbeat within 2x interval)
          const HEARTBEAT_TIMEOUT_MS = 60000 // 60 seconds (2x 30s interval)
          const now = Date.now()
          if (this.lastHeartbeatTime && (now - this.lastHeartbeatTime) < HEARTBEAT_TIMEOUT_MS) {
            checks.heartbeat = 'publishing'
          } else {
            checks.heartbeat = 'stopped'
            isHealthy = false
          }
        }

        return {
          status: isHealthy ? 'healthy' : 'unhealthy',
          agentId: this.config.agentId,
          timestamp: new Date().toISOString(),
          checks
        }
      }
    }

    Export function createHealthCheckServer(config: HealthCheckConfig): HealthCheckServer {
      return new HealthCheckServer(config)
    }
