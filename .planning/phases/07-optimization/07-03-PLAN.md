<objective>
Implement context reference passing for large payloads to reduce bandwidth and improve throughput through hash-based deduplication.

Purpose: When task contexts (file contents, codebases, large data) exceed 10KB, store them in SQLite with SHA-256 hash as key and pass only the hash reference via MQTT. This reduces network traffic by 60-80% for large payloads and enables automatic deduplication.

Output: ContextManager class with SQLite storage, SHA-256 hash-based reference passing, and automatic deduplication.

```typescript
export function initializeSchema(db: Database.Database): void {

  db.exec(`
    CREATE TABLE IF NOT EXISTS context_refs (
      hash BLOB NOT NULL PRIMARY KEY,
      size INTEGER NOT NULL,
      content BLOB NOT NULL,
      created_at INTEGER NOT NULL DEFAULT (strftime('%s', 'now')),
      access_count INTEGER DEFAULT 1,
      last_accessed INTEGER NOT NULL DEFAULT (strftime('%s', 'now'))
    ) WITHOUT ROWID;

    CREATE INDEX IF NOT EXISTS idx_context_refs_accessed ON context_refs(last_accessed);
    CREATE INDEX IF NOT EXISTS idx_context_refs_created ON context_refs(created_at);
  `);
}

export function createGarbageCollectionQuery(db: Database.Database) {
  // Delete contexts unused > 7 days OR with low access count
  return db.prepare(`
    DELETE FROM context_refs
    WHERE last_accessed < strftime('%s', 'now', '-7 days')
       OR (access_count < 3 AND created_at < strftime('%s', 'now', '-3 days'))
  `);
}
```

## Schema Design

Per RESEARCH.md Pattern 3: Hash-based deduplication using WITHOUT ROWID optimization:

- **hash**: SHA-256 hash (32 bytes) as primary key - automatic deduplication
- **size**: Content size in bytes - for metrics
- **content**: Actual context payload (BLOB) - compressed if needed
- **created_at**: Unix timestamp - for retention policy
- **access_count**: Number of times referenced - for garbage collection heuristics
- **last_accessed**: Unix timestamp - for LRU garbage collection

WITHOUT ROWID optimization: hash IS the row identifier, no additional rowid column needed (per Pwned Passwords pattern).

## Garbage Collection Policy

Per RESEARCH.md Open Question 2: Start with 7-day TTL, monitor usage, adjust based on disk space.

Delete contexts that are:
- Unused > 7 days (last_accessed < 7 days ago)
- OR rarely used AND old (access_count < 3 AND created_at < 3 days ago)

This keeps frequently-used contexts while cleaning up old/unused ones.

```typescript
// packages/coordination/src/optimization/context-manager.ts

import Database from 'better-sqlite3';
import * as crypto from 'crypto';
import * as fs from 'fs';

export interface ContextReference {
  ref: string;        // SHA-256 hash as hex string
  size: number;       // Original size in bytes
  compressed: boolean; // Whether content is compressed
}

export interface ContextOptions {
  thresholdBytes?: number;  // Size threshold for reference passing (default: 10KB)
  enableCompression?: boolean; // Compress content before storage
  retentionDays?: number;  // Garbage collection retention (default: 7)
}

export class ContextManager {
  private db: Database.Database;
  private options: Required<ContextOptions>;

  constructor(db: Database.Database, options: ContextOptions = {}) {
    this.db = db;
    this.options = {
      thresholdBytes: options.thresholdBytes || 10240, // 10KB per OPTI-05
      enableCompression: options.enableCompression || false,
      retentionDays: options.retentionDays || 7
    };
  }

  /**
   * Store context content and return reference if large enough
   */
  storeContext(content: Buffer): ContextReference | null {
    // Only store if above threshold
    if (content.length < this.options.thresholdBytes) {
      return null;
    }

    const hash = crypto.createHash('sha256').update(content).digest();

    // Check for existing (deduplication)
    const existing = this.db.prepare(
      'SELECT hash FROM context_refs WHERE hash = ?'
    ).get(hash) as { hash: Buffer } | undefined;

    if (existing) {
      // Update access count and last_accessed
      this.db.prepare(`
        UPDATE context_refs
        SET access_count = access_count + 1,
            last_accessed = strftime('%s', 'now')
        WHERE hash = ?
      `).run(hash);

      return {
        ref: hash.toString('hex'),
        size: content.length,
        compressed: false
      };
    }

    // Store new context
    const storeContent = this.options.enableCompression
      ? this.compressContent(content)
      : content;

    this.db.prepare(`
      INSERT INTO context_refs (hash, size, content)
      VALUES (?, ?, ?)
    `).run(hash, content.length, storeContent);

    return {
      ref: hash.toString('hex'),
      size: content.length,
      compressed: this.options.enableCompression
    };
  }

  /**
   * Retrieve context content by reference hash
   */
  getContext(ref: string): Buffer | null {
    const hash = Buffer.from(ref, 'hex');

    const row = this.db.prepare(`
      SELECT content, size
      FROM context_refs
      WHERE hash = ?
    `).get(hash) as { content: Buffer; size: number } | undefined;

    if (!row) {
      return null;
    }

    // Update access tracking
    this.db.prepare(`
      UPDATE context_refs
      SET access_count = access_count + 1,
          last_accessed = strftime('%s', 'now')
      WHERE hash = ?
    `).run(hash);

    // Decompress if needed
    if (this.options.enableCompression) {
      return this.decompressContent(row.content);
    }

    return row.content;
  }

  /**
   * Check if content should use reference passing
   */
  shouldUseReference(content: Buffer): boolean {
    return content.length >= this.options.thresholdBytes;
  }

  /**
   * Run garbage collection to clean up old contexts
   */
  runGarbageCollection(): number {
    const stmt = this.db.prepare(`
      DELETE FROM context_refs
      WHERE last_accessed < strftime('%s', 'now', '-${this.options.retentionDays} days')
         OR (access_count < 3 AND created_at < strftime('%s', 'now', '-3 days'))
    `);

    const result = stmt.run();
    return result.changes;
  }

  /**
   * Get context storage statistics
   */
  getStats() {
    const totalContexts = this.db.prepare(
      'SELECT COUNT(*) as count FROM context_refs'
    ).get() as { count: number };

    const totalSize = this.db.prepare(
      'SELECT SUM(size) as total FROM context_refs'
    ).get() as { total: number };

    return {
      totalContexts: totalContexts.count,
      totalBytes: totalSize.total || 0,
      thresholdBytes: this.options.thresholdBytes,
      retentionDays: this.options.retentionDays
    };
  }

  private compressContent(content: Buffer): Buffer {
    // Placeholder for compression (can use zlib in future)
    // For now, return uncompressed
    return content;
  }

  private decompressContent(content: Buffer): Buffer {
    // Placeholder for decompression
    return content;
  }
}
```

```typescript
// packages/coordination/src/optimization/context-manager.ts (add to file)

export async function prepareMessagePayload(
  payload: any,
  contextManager: ContextManager
): Promise<any> {
  // Check if payload has large context content
  if (payload?.context?.content) {
    const content = Buffer.isBuffer(payload.context.content)
      ? payload.context.content
      : Buffer.from(payload.context.content);

    // Use reference passing if large enough
    if (contextManager.shouldUseReference(content)) {
      const ref = contextManager.storeContext(content);
      if (ref) {
        // Replace content with reference
        return {
          ...payload,
          context: {
            ref: ref,
            content: undefined
          }
        };
      }
    }
  }

  return payload;
}

export async function resolveMessagePayload(
  payload: any,
  contextManager: ContextManager
): Promise<any> {
  // Check if payload has context reference
  if (payload?.context?.ref) {
    const content = contextManager.getContext(payload.context.ref);
    if (content) {
      // Resolve reference to actual content
      return {
        ...payload,
        context: {
          content: content.toString(),
          ref: undefined
        }
      };
    }
  }

  return payload;
}
```

These utilities automate context reference passing:
- prepareMessagePayload(): Check content size, store if large, replace with ref
- resolveMessagePayload(): Detect ref, retrieve content, replace ref with content
